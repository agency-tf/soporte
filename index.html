Sincronización Bidireccional Activa

Esta plantilla de Apps Script está diseñada para mantener en sincronía la información de Transacciones y Arqueos de Caja en Google Sheets. Si es eliminado un registro en la hoja Excel o el portal web, el sistema reajustará las bases automáticamente.

Código Apps Script Requerido 
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var payload = JSON.parse(e.postData.contents);
    var type = payload.type;
    var data = payload.data;
    
    if (type === 'transaction') {
      var sheet = ss.getSheetByName("Transacciones");
      if (!sheet) return createResponse({ status: "error", msg: "Hoja Transacciones no existe" });
      var amt = parseFloat(data.amount);
      if (data.type === 'retiro') {
        amt = -Math.abs(amt);
      } else {
        amt = Math.abs(amt);
      }
      sheet.appendRow([
        data.id,
        data.date,
        data.type,
        amt,
        data.movementNumber || "S/N",
        data.concept || "Sin detalle",
        data.user,
        new Date()
      ]);
      return createResponse({ status: "success", msg: "Transacción guardada" });
    }
    
    if (type === 'reconcile') {
      var sheet = ss.getSheetByName("Cuadres");
      if (!sheet) return createResponse({ status: "error", msg: "Hoja Cuadres no existe" });
      sheet.appendRow([
        data.id,
        data.date,
        data.user,
        data.expected,
        data.physical,
        data.difference,
        data.observations || "",
        data.status || "Pendiente",
        data.auditedBy || "N/A"
      ]);
      return createResponse({ status: "success", msg: "Cuadre guardado" });
    }
    
    if (type === 'delete') {
      var idToDelete = payload.id;
      var sheetT = ss.getSheetByName("Transacciones");
      var sheetC = ss.getSheetByName("Cuadres");
      var deleted = false;
      
      if (sheetT) {
        var rowsT = sheetT.getDataRange().getValues();
        for (var i = 1; i < rowsT.length; i++) {
          if (rowsT[i][0] === idToDelete) {
            sheetT.deleteRow(i + 1);
            deleted = true;
            break;
          }
        }
      }
      if (!deleted && sheetC) {
        var rowsC = sheetC.getDataRange().getValues();
        for (var j = 1; j < rowsC.length; j++) {
          if (rowsC[j][0] === idToDelete) {
            sheetC.deleteRow(j + 1);
            deleted = true;
            break;
          }
        }
      }
      return createResponse({ status: "success", msg: deleted ? "Eliminado" : "No encontrado" });
    }
  } catch (err) {
    return createResponse({ status: "error", msg: err.toString() });
  }
}

function doGet() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetT = ss.getSheetByName("Transacciones");
    var sheetC = ss.getSheetByName("Cuadres");
    var tx = [];
    var rec = [];
    
    if (sheetT) {
      var dataT = sheetT.getDataRange().getValues();
      for (var i = 1; i < dataT.length; i++) {
        var amt = parseFloat(dataT[i][3]);
        var type = dataT[i][2];
        if (amt < 0) {
          type = 'retiro';
          amt = Math.abs(amt);
        }
        tx.push({
          id: dataT[i][0],
          date: dataT[i][1],
          type: type,
          amount: amt,
          movementNumber: dataT[i][4],
          concept: dataT[i][5],
          user: dataT[i][6]
        });
      }
    }
    
    if (sheetC) {
      var dataC = sheetC.getDataRange().getValues();
      for (var j = 1; j < dataC.length; j++) {
        rec.push({
          id: dataC[j][0],
          date: dataC[j][1],
          user: dataC[j][2],
          expected: parseFloat(dataC[j][3]),
          physical: parseFloat(dataC[j][4]),
          difference: parseFloat(dataC[j][5]),
          observations: dataC[j][6],
          status: dataC[j][7],
          auditedBy: dataC[j][8]
        });
      }
    }
    return createResponse({ status: "success", transactions: tx, reconciliations: rec });
  } catch (err) {
    return createResponse({ status: "error", msg: err.toString() });
  }
}

function createResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
