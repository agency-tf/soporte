GoogleAppsScript.js 
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var payload = JSON.parse(e.postData.contents);
    
    // Si la hoja está totalmente en blanco, colocar los encabezados para el banco
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["ID Transacción", "Fecha", "Tipo", "Monto", "Movimiento Nº", "Concepto", "Usuario Cajero", "Fecha de Sincronización"]);
    }
    
    // Guardar los datos recibidos (Sincronización Total)
    if (Array.isArray(payload)) {
      payload.forEach(function(item) {
        sheet.appendRow([
          item.id,
          item.date,
          item.type,
          item.amount,
          item.movementNumber || "S/N",
          item.concept || "Sin detalle",
          item.user,
          new Date()
        ]);
      });
    } else {
      // Auto-sincronización de un solo elemento
      sheet.appendRow([
        payload.id,
        payload.date,
        payload.type,
        payload.amount,
        payload.movementNumber || "S/N",
        payload.concept || "Sin detalle",
        payload.user,
        new Date()
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Sincronizado" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet() {
  return ContentService.createTextOutput(JSON.stringify({ status: "online", active: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
